#!/usr/bin/env python3
"""
FM Portal Login Integration Script

This script generates an HTML file with a login form that submits to the FM Portal.
The generated file can be integrated into an existing website under an "insured tab" function.

Usage:
1. Run this script in Replit
2. It will create a file called 'fm_portal_login.html'
3. Follow the integration instructions printed at the end
"""

import os

    Creates an HTML file with a login form that submits to the FM Portal
    """
    html_content = """<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>FM Portal Login</title>
    <style>
        body {
            font-family: Arial, sans-serif;

        }
        .login-container {

            margin: 0 auto;
            background: white;

            border-radius: 5px;
            box-shadow: 0 0 10px rgba(0, 0, 0, 0.1);
        }
        h2 {
            text-align: center;
            color: #333;
        }
        .form-group {
            margin-bottom: 15px;
        }
        label {
            display: block;
            margin-bottom: 5px;
            font-weight: bold;
        }
        input[type="text"],
        input[type="password"] {
            width: 100%;

    with open('fm_portal_login.html', 'w') as file:

    print(f"Created FM Portal login HTML file: {os.path.abspath('fm_portal_login.html')}")

def print_integration_instructions():
    """
    Prints instructions for integrating the login form into an existing website
    """
    instructions = """
INTEGRATION INSTRUCTIONS:
========================

Option 1: Direct Link
--------------------
Add a link to the FM Portal login page in your website's navigation menu:

<li><a href="fm_portal_login.html">Insured Login</a></li>

Option 2: Embed in Tab System
----------------------------
If your website uses a tab system, you can integrate the login form as follows:

1. Add a new tab button in your HTML:
   <button class="tablinks" onclick="openTab(event, 'InsuredLogin')">Insured Login</button>

2. Add the tab content div:
   <div id="InsuredLogin" class="tabcontent">
     <!-- Include the FM Portal login form here -->
     <iframe src="fm_portal_login.html" width="100%" height="400" frameborder="0"></iframe>
   </div>

3. Make sure you have the necessary JavaScript for tab functionality:
   <script>
   function openTab(evt, tabName) {
     var i, tabcontent, tablinks;

     // Hide all tab content
     tabcontent = document.getElementsByClassName("tabcontent");
     for (i = 0; i < tabcontent.length; i++) {
       tabcontent[i].style.display = "none";
     }

     // Remove "active" class from all tab buttons
     tablinks = document.getElementsByClassName("tablinks");
     for (i = 0; i < tablinks.length; i++) {
       tablinks[i].className = tablinks[i].className.replace(" active", "");
     }

     // Show the current tab and add "active" class to the button
     document.getElementById(tabName).style.display = "block";
     evt.currentTarget.className += " active";
   }
   </script>

Option 3: Direct HTML Integration
-------------------------------
Instead of using the generated HTML file, you can copy the HTML content directly into your website's code.
Look for the <div class="login-container"> section in the generated HTML file and copy it to the appropriate
location in your website's HTML.

Note: Make sure to also copy the CSS styles from the <style> section or include them in your website's stylesheet.
"""
    print(instructions)

def main():
    """
    Main function to create the FM Portal login HTML file and print integration instructions
    """
    create_fm_portal_login_html()
    print_integration_instructions()

if __name__ == "__main__":
    main()
